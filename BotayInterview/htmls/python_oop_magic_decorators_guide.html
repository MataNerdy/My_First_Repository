Инкапсуляция

class BankAccount:
    def __init__(self, balance: float = 0):
        self._balance = balance

    @property
    def balance(self) -> float:
        return self._balance

    def withdraw(self, amount: float) -> None:
        if amount <= 0:
            raise ValueError("Сумма должна быть положительной")
        if amount > self._balance:
            raise ValueError("Недостаточно средств")

        self._balance -= amount
Подчёркивания в Python _name — соглашение «внутренняя деталь». __name — name mangling, а не настоящая приватность.

__repr__ и __str__

Два строковых представления для разных аудиторий.

__str__

Для пользователя. Читаемый, удобный вывод.

__repr__

Для разработчика. Точный, диагностический вывод.

class Candidate:
    def __init__(self, name: str, role: str):
        self.name = name
        self.role = role

    def __str__(self) -> str:
        return f"{self.name} — кандидат на {self.role}"

    def __repr__(self) -> str:
        return (
            f"Candidate("
            f"name={self.name!r}, "
            f"role={self.role!r}"
            f")"
        )
ВыражениеИспользует

Безопасность Не помещай в __repr__ пароли, токены, API-ключи и другие секреты: представление часто попадает в логи и traceback.

__eq__, is и __hash__

Равенство значений, идентичность и контракт хеширования.

==

Семантика равенства, заданная типом.

is

Один ли это объект в памяти. Не переопределяется.

class Candidate:
    def __init__(self, name: str, role: str):
        self.name = name
        self.role = role

    def __eq__(self, other: object) -> bool:
        if not isinstance(other, Candidate):
            return NotImplemented

        return (self.name, self.role) == (other.name, other.role)

Почему NotImplemented, а не False?

False означает «я сравнил и объекты не равны». NotImplemented означает «этот тип сравнения мне неизвестен», после чего Python может попробовать обратную реализацию второго объекта.

Не путать NotImplemented — специальное значение. NotImplementedError — исключение.

is None

if value is None:
    ...

# а не:
if value == None:
    ...

== может быть переопределён, is — нет.

Контракт хеширования

Обязательное правило Если a == b, то должно быть hash(a) == hash(b). 
class ImmutableCandidate:
    def __init__(self, name: str, role: str):
        self.name = name
        self.role = role

    def __eq__(self, other: object) -> bool:
        if not isinstance(other, ImmutableCandidate):
            return NotImplemented
        return (self.name, self.role) == (other.name, other.role)

    def __hash__(self) -> int:
        return hash((self.name, self.role))

Поля, влияющие на хеш, нельзя изменять после помещения объекта в словарь или множество. Поэтому хешируемые value-объекты обычно делают неизменяемыми.

Отдельный класс итератора

class InclusiveRangeIterator:
    def __init__(self, start: int, stop: int):
        self._current = start
        self._stop = stop

    def __iter__(self) -> "InclusiveRangeIterator":
        return self

    def __next__(self) -> int:
        if self._current > self._stop:
            raise StopIteration

        value = self._current
        self._current += 1
        return value


class InclusiveRange:
    def __init__(self, start: int, stop: int):
        self.start = start
        self.stop = stop

    def __iter__(self) -> InclusiveRangeIterator:
        return InclusiveRangeIterator(self.start, self.stop)
Правильное разделение Коллекция создаёт новый итератор для каждого обхода. Итератор хранит состояние одного конкретного обхода.
Генераторный вариант 
from collections.abc import Iterator


class SkillSet:
    def __init__(self, skills: list[str]):
        self._skills = list(skills)

    def __iter__(self) -> Iterator[str]:
        yield from self._skills

import time
from types import TracebackType


class Timer:
    def __init__(self, name: str):
        self.name = name
        self.elapsed: float | None = None
        self._start: float | None = None

    def __enter__(self) -> "Timer":
        self._start = time.perf_counter()
        return self

    def __exit__(
        self,
        exc_type: type[BaseException] | None,
        exc_value: BaseException | None,
        traceback: TracebackType | None,
    ) -> bool:
        if self._start is None:
            raise RuntimeError("Таймер не был запущен")

        self.elapsed = time.perf_counter() - self._start
        print(f"{self.name}: {self.elapsed:.6f} секунд")

        return False

Механика декораторов

Декоратор принимает функцию и возвращает новую вызываемую сущность.

Синтаксический сахар @logger над функцией означает func = logger(func). 
def logger(function):
    def wrapper():
        print("До вызова")
        result = function()
        print("После вызова")
        return result

    return wrapper


@logger
def say_hello() -> str:
    return "Привет!"
исходная функция

Когда выполняется декоратор

  • внешняя функция-декоратор выполняется при определении функции, обычно при импорте модуля;
  • обёртка выполняется при каждом вызове декорированной функции.
Вопрос

Почему в конце декоратора нужно писать return wrapper, а не return wrapper()?

return wrapper возвращает сам объект-функцию. return wrapper() немедленно вызовет обёртку во время декорирования и вернёт результат её работы.

Универсальный декоратор

*args, **kwargs, возврат результата и functools.wraps.

*args и **kwargs

Базовый шаблон

from functools import wraps


def decorator(function):
    @wraps(function)
    def wrapper(*args, **kwargs):
        # действия до вызова

        result = function(*args, **kwargs)

        # действия после вызова

        return result

    return wrapper
Три обязательные вещи Обёртка принимает *args, **kwargs, передаёт их как function(*args, **kwargs) и возвращает результат.

Зачем functools.wraps

Без него декорированная функция теряет имя, документацию, аннотации и удобную introspection-информацию.

from functools import wraps


def logger(function):
    @wraps(function)
    def wrapper(*args, **kwargs):
        print(f"Вызов {function.__name__}")
        return function(*args, **kwargs)

    return wrapper

Таймер, работающий и при исключении

import time
from functools import wraps


def timer(function):
    @wraps(function)
    def wrapper(*args, **kwargs):
        start = time.perf_counter()

        try:
            return function(*args, **kwargs)
        finally:
            elapsed = time.perf_counter() - start
            print(
                f"{function.__name__}: "
                f"{elapsed:.6f} секунд"
            )

    return wrapper

Счётчик вызовов

from functools import wraps


def count_calls(function):
    count = 0

    @wraps(function)
    def wrapper(*args, **kwargs):
        nonlocal count
        count += 1

        print(f"{function.__name__}: вызов №{count}")
        return function(*args, **kwargs)

    return wrapper

Точная типизация

from collections.abc import Callable
from functools import wraps
from typing import ParamSpec, TypeVar


P = ParamSpec("P")
R = TypeVar("R")


def logger(
    function: Callable[P, R],
) -> Callable[P, R]:

    @wraps(function)
    def wrapper(
        *args: P.args,
        **kwargs: P.kwargs,
    ) -> R:
        print(f"Вызов {function.__name__}")
        return function(*args, **kwargs)

    return wrapper
Не возвращай значение из finally return внутри finally способен перезаписать нормальный результат и даже подавить исключение.

Устные ответы для собеседования

Формулировки на 30–60 секунд.

Класс и объект
Класс описывает структуру и поведение объектов: атрибуты, методы и специальные протоколы. Объект — конкретный экземпляр класса со своим состоянием. Разные экземпляры используют методы одного класса, но хранят разные значения атрибутов.
Инкапсуляция, наследование и полиморфизм
Инкапсуляция объединяет данные и методы и позволяет классу контролировать корректность своего состояния. Наследование создаёт новый класс на основе существующего и позволяет переиспользовать или переопределять поведение. Полиморфизм позволяет работать с разными объектами через общий интерфейс; в Python он часто основан на duck typing.
self
self — ссылка на конкретный экземпляр. При вызове obj.method() Python автоматически связывает объект с методом и передаёт его первым аргументом. self не является ключевым словом, но это стандартное имя.
__repr__ и __str__
__str__ отвечает за читаемый пользовательский вывод и используется функцией str и обычно print. __repr__ предназначен для разработчика, отладки и представления объектов внутри контейнеров. Если __str__ не реализован, Python может использовать __repr__ как запасной вариант.
__eq__, is и NotImplemented
__eq__ задаёт семантику оператора ==. При неподдерживаемом типе корректно вернуть NotImplemented, чтобы Python мог попробовать обратное сравнение. is не сравнивает значения, а проверяет, являются ли две ссылки одним объектом; основной практический случай — is None.
Iterable и iterator
Итерируемый объект позволяет получить итератор через iter. Итератор хранит состояние обхода, реализует __next__ и выдаёт следующий элемент или бросает StopIteration. Цикл for получает итератор и вызывает next до завершения.
__new__ и __init__
__new__ создаёт и возвращает объект, а __init__ инициализирует уже созданный экземпляр. __new__ особенно нужен при наследовании от неизменяемых типов или контроле создания экземпляров. __init__ обязан возвращать None.
Контекстный менеджер
__enter__ подготавливает ресурс и возвращает значение после as. __exit__ вызывается при выходе из успешно открытого контекста независимо от нормального завершения или исключения, поэтому в нём освобождают ресурс. Истинное значение из __exit__ подавляет исключение.
Замыкание и декоратор
Замыкание — функция, сохраняющая доступ к переменным внешней функции после её завершения. Декоратор принимает функцию и возвращает новую функцию или другой вызываемый объект. Обёртка хранит исходную функцию в замыкании и добавляет поведение до или после вызова. Запись @decorator эквивалентна func = decorator(func).

Задача 1. Класс-коллекция SkillSet

Реализовать __len__, __contains__, __iter__ и __repr__.

from collections.abc import Iterator


class SkillSet:
    def __init__(self, skills: list[str]):
        self._skills = list(skills)

    def __len__(self) -> int:
        return len(self._skills)

    def __contains__(self, skill: object) -> bool:
        return skill in self._skills

    def __iter__(self) -> Iterator[str]:
        return iter(self._skills)

    def __repr__(self) -> str:
        return f"SkillSet(skills={self._skills!r})"
Задача 2. ProfileBlock

Контекстный менеджер, накапливающий длительности нескольких блоков.

import time
from types import TracebackType


class ProfileBlock:
    def __init__(self, name: str):
        self.name = name
        self._start: float | None = None
        self._durations: list[float] = []

    def __enter__(self) -> "ProfileBlock":
        if self._start is not None:
            raise RuntimeError("Профилировщик уже используется")

        self._start = time.perf_counter()
        return self

    def __exit__(
        self,
        exc_type: type[BaseException] | None,
        exc_value: BaseException | None,
        traceback: TracebackType | None,
    ) -> bool:
        if self._start is None:
            raise RuntimeError("Профилирование не запущено")

        self._durations.append(
            time.perf_counter() - self._start
        )
        self._start = None
        return False

    def get_stats(self) -> dict[str, int | float]:
        if not self._durations:
            return {
                "count": 0,
                "min": 0.0,
                "max": 0.0,
                "avg": 0.0,
            }

        return {
            "count": len(self._durations),
            "min": min(self._durations),
            "max": max(self._durations),
            "avg": sum(self._durations) / len(self._durations),
        }
Задача 3. Декоратор timer

Сохранить сигнатуру, вернуть результат и измерить время даже при исключении.

import time
from collections.abc import Callable
from functools import wraps
from typing import ParamSpec, TypeVar


P = ParamSpec("P")
R = TypeVar("R")


def timer(
    function: Callable[P, R],
) -> Callable[P, R]:

    @wraps(function)
    def wrapper(
        *args: P.args,
        **kwargs: P.kwargs,
    ) -> R:
        start = time.perf_counter()

        try:
            return function(*args, **kwargs)
        finally:
            elapsed = time.perf_counter() - start
            print(
                f"{function.__name__}: "
                f"{elapsed:.6f} секунд"
            )

    return wrapper
